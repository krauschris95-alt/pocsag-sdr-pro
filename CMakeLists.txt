cmake_minimum_required(VERSION 3.24)
project(POCSAG_SDR_Pro VERSION 0.1.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(pocsag_core src/decoder.cpp)
target_include_directories(pocsag_core PUBLIC src)
enable_testing()
add_executable(decoder_tests tests/decoder_tests.cpp)
target_link_libraries(decoder_tests PRIVATE pocsag_core)
add_test(NAME decoder COMMAND decoder_tests)
add_executable(pocsag-iq-decode tests/iq_decode.cpp)
target_link_libraries(pocsag-iq-decode PRIVATE pocsag_core)
option(BUILD_GUI "Build Qt desktop application" ON)
if(BUILD_GUI)
 find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets Sql)
 qt_standard_project_setup()
 qt_add_executable(POCSAG-SDR-Pro WIN32 src/main.cpp src/window.cpp src/receiver.cpp src/options.cpp src/configuration.cpp)
 target_link_libraries(POCSAG-SDR-Pro PRIVATE pocsag_core Qt6::Widgets Qt6::Sql)
 install(TARGETS POCSAG-SDR-Pro RUNTIME DESTINATION .)
 add_executable(integration_tests tests/integration_tests.cpp src/window.cpp src/receiver.cpp src/options.cpp src/configuration.cpp)
 target_link_libraries(integration_tests PRIVATE pocsag_core Qt6::Widgets Qt6::Sql)
 add_test(NAME integration COMMAND integration_tests -platform offscreen)
 add_executable(reference_tests tests/reference_tests.cpp)
 target_link_libraries(reference_tests PRIVATE pocsag_core Qt6::Core)
 target_compile_definitions(reference_tests PRIVATE FIXTURE_DIRECTORY="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
 add_test(NAME independent_reference COMMAND reference_tests)
endif()
