cmake_minimum_required(VERSION 3.24)
project(POCSAG_SDR_Pro VERSION 0.1.16 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(WIN32 AND CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
 add_compile_options(-march=x86-64 -mtune=generic)
endif()
add_library(pocsag_core src/decoder.cpp src/payloadanalysis.cpp)
target_include_directories(pocsag_core PUBLIC src)
enable_testing()
add_executable(decoder_tests tests/decoder_tests.cpp)
target_link_libraries(decoder_tests PRIVATE pocsag_core)
add_test(NAME decoder COMMAND decoder_tests)
add_executable(pocsag-iq-decode tests/iq_decode.cpp)
target_link_libraries(pocsag-iq-decode PRIVATE pocsag_core)
option(BUILD_GUI "Build Qt desktop application" ON)
if(BUILD_GUI)
 find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets Sql Network)
 qt_standard_project_setup()
 add_library(sdr_adapters src/soapyradio.cpp)
 target_include_directories(sdr_adapters PUBLIC src PRIVATE third_party/SoapySDR)
 target_link_libraries(sdr_adapters PUBLIC Qt6::Core)
 add_library(desktop_services src/speech.cpp src/journalworker.cpp src/pagerworker.cpp src/pagerhttps.cpp)
 target_link_libraries(desktop_services PUBLIC Qt6::Core Qt6::Sql Qt6::Network)
 if(WIN32)
  target_link_libraries(desktop_services PRIVATE sapi ole32 uuid)
 endif()
 add_library(desktop_window src/window.cpp src/presentation.cpp src/frequencydialog.cpp src/receiver.cpp src/options.cpp src/configuration.cpp src/speechoptions.cpp src/desktoplayout.cpp src/receiveroptions.cpp src/multiradio.cpp src/ingestion.cpp src/txtprofiles.cpp src/datastore.cpp src/pageroptions.cpp)
 if(WIN32)
 target_link_libraries(desktop_window PRIVATE dwmapi)
 endif()
 target_link_libraries(desktop_window PUBLIC pocsag_core desktop_services sdr_adapters Qt6::Widgets Qt6::Sql)
 qt_add_executable(POCSAG-SDR-Pro WIN32 src/main.cpp)
 target_link_libraries(POCSAG-SDR-Pro PRIVATE desktop_window)
 install(TARGETS POCSAG-SDR-Pro RUNTIME DESTINATION .)
 add_executable(integration_tests tests/integration_tests.cpp)
 target_link_libraries(integration_tests PRIVATE desktop_window)
 target_compile_definitions(integration_tests PRIVATE FIXTURE_DIRECTORY="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
 add_executable(soapy_probe tests/soapy_probe.cpp src/receiver.cpp)
 target_link_libraries(soapy_probe PRIVATE pocsag_core sdr_adapters Qt6::Core)
 add_test(NAME integration COMMAND integration_tests -platform offscreen)
 add_executable(reference_tests tests/reference_tests.cpp)
 target_link_libraries(reference_tests PRIVATE pocsag_core Qt6::Core)
 target_compile_definitions(reference_tests PRIVATE FIXTURE_DIRECTORY="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
 add_test(NAME independent_reference COMMAND reference_tests)
 add_executable(speech_tests tests/speech_tests.cpp)
 target_include_directories(speech_tests PRIVATE src)
 target_link_libraries(speech_tests PRIVATE desktop_services)
 add_test(NAME speech_queue COMMAND speech_tests)
 add_executable(speech_probe tests/speech_probe.cpp)
 target_include_directories(speech_probe PRIVATE src)
 target_link_libraries(speech_probe PRIVATE desktop_services)
 if(WIN32)
  add_library(mock_rtl SHARED tests/mock_rtl.cpp)
  set_target_properties(mock_rtl PROPERTIES PREFIX "")
  add_executable(receiver_tests tests/receiver_tests.cpp src/receiver.cpp)
  target_link_libraries(receiver_tests PRIVATE pocsag_core sdr_adapters Qt6::Core)
  target_compile_definitions(receiver_tests PRIVATE FIXTURE_DIRECTORY="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
  add_dependencies(receiver_tests mock_rtl)
  add_test(NAME receiver_reconnect_injected COMMAND receiver_tests)
 endif()
endif()
if(BUILD_GUI)
 add_executable(capcode_regression_tests tests/capcode_regression_tests.cpp)
 target_link_libraries(capcode_regression_tests PRIVATE desktop_window)
 add_test(NAME capcode_save_regression COMMAND capcode_regression_tests -platform offscreen)

 else()
